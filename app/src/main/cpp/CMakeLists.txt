cmake_minimum_required(VERSION 3.22.1)
set(CMAKE_BUILD_TYPE Release CACHE STRING "" FORCE)
project(lusifer_llm LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_C_STANDARD 11)

set(LLAMA_DIR ${CMAKE_CURRENT_SOURCE_DIR}/llama.cpp)
add_library(lusifer_llm SHARED llm_jni.cpp)
find_library(log-lib log)

if(EXISTS ${LLAMA_DIR}/CMakeLists.txt)
    set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
    set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
    set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
    set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
    set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
    set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
    set(LLAMA_CURL OFF CACHE BOOL "" FORCE)
    set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
    set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
    set(GGML_LLAMAFILE OFF CACHE BOOL "" FORCE)
    add_subdirectory(${LLAMA_DIR} llama_build)
    target_compile_definitions(lusifer_llm PRIVATE HAVE_LLAMA=1)
    target_include_directories(lusifer_llm PRIVATE ${LLAMA_DIR}/include ${LLAMA_DIR}/ggml/include)
    target_link_libraries(lusifer_llm llama ${log-lib})
else()
    message(WARNING "llama.cpp bulunamadi: LLM devre disi (stub) derleniyor.")
    target_link_libraries(lusifer_llm ${log-lib})
endif()

target_link_options(lusifer_llm PRIVATE -Wl,-z,max-page-size=16384)
